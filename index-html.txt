<script>
    document.addEventListener('DOMContentLoaded', function() {
        const tabs = document.querySelectorAll('.tab-btn');
        const contents = {
            education: document.getElementById('education'),
            scholarship: document.getElementById('scholarship'),
            business: document.getElementById('business')
        };
        tabs.forEach(btn => {
            btn.addEventListener('click', function() {
                tabs.forEach(b => b.classList.remove('active'));
                this.classList.add('active');
                Object.values(contents).forEach(c => c.classList.remove('active'));
                const target = this.dataset.tab;
                if (contents[target]) contents[target].classList.add('active');
            });
        });
        calculateFreight();
    });

    function calculateFreight() {
        const cat = document.getElementById('productCat').value;
        const weight = parseFloat(document.getElementById('weightKg').value) || 0;
        const mode = document.getElementById('shipMode').value;
        let ratePerKg = 0;
        switch(cat) {
            case 'electronics': ratePerKg = 8.5; break;
            case 'clothing': ratePerKg = 6.0; break;
            case 'machinery': ratePerKg = 7.0; break;
            case 'general': ratePerKg = 5.5; break;
            default: ratePerKg = 6.5;
        }
        let multiplier = (mode === 'air') ? 1.8 : 0.6;
        let total = (weight * ratePerKg * multiplier) + 50;
        if (total < 0) total = 0;
        document.getElementById('estimatedPrice').textContent = '$' + total.toFixed(2);
        document.getElementById('freightEstimate').value = total.toFixed(2);
        document.getElementById('formFreightEstimate').value = total.toFixed(2);
    }
    document.querySelectorAll('#productCat, #weightKg, #shipMode').forEach(el => {
        el.addEventListener('change', calculateFreight);
        el.addEventListener('input', calculateFreight);
    });
</script>
